# accounts.hynt.one — the Hynt identity provider.
# Pins are deliberate: this service issues the tokens every other platform
# trusts, so its dependency set changes only on purpose.

# --- web (Rust-powered; HTTP, routing and JSON never enter Python) ---
cello-framework==1.3.0

# --- database ---
# Sync SQLAlchemy on psycopg2. Cello runs sync handlers on a bounded blocking
# threadpool (`blocking=True`), so a sync driver is the simpler correct choice
# here — there is no asyncio loop to starve.
sqlalchemy>=2.0.36,<2.1
psycopg2-binary==2.9.10
alembic==1.14.0

# --- validation / settings ---
pydantic==2.10.4
pydantic-settings==2.7.0

# --- crypto ---
# RS256 token signing. Every platform verifies locally against the published
# JWKS, so a request on terminal.hynt.one never calls back here.
pyjwt[crypto]==2.10.1
cryptography>=44.0.0
# Argon2id for passwords — the three platforms each rolled their own
# scrypt/pbkdf2; this is the one place that decision now lives.
argon2-cffi==23.1.0

python-dotenv==1.0.1

# --- tests ---
pytest==8.3.4
httpx==0.28.1
